// XAU/USD 24/24 GOLD NEWS & SESSION TELEGRAM MONITOR (FREE)
// Run on: https://script.google.com (Triggers -> Every 15 mins)
// ==========================================================
const TELEGRAM_TOKEN = "YOUR_BOT_TOKEN_HERE";
const TELEGRAM_CHAT_ID = "YOUR_CHAT_ID_HERE";

function checkGoldMarket247() {
  const props = PropertiesService.getScriptProperties();
  const now = new Date();
  const khHour = Number(Utilities.formatDate(now, "Asia/Phnom_Penh", "HH"));
  const khMin = Number(Utilities.formatDate(now, "Asia/Phnom_Penh", "mm"));

  // 1. Alert before London (14:45) & New York (19:45) Gold Sessions
  if ((khHour === 14 || khHour === 19) && khMin < 15) {
    const sessionName = khHour === 14 ? "London Session (3:00 PM)" : "New York Peak Session (8:00 PM)";
    sendTgMessage(`🔔 ម៉ោងទីផ្សារមាសសកម្មជិតចាប់ផ្តើម!\n⏰ ម៉ោងកម្ពុជា៖ ${khHour}:${khMin < 10 ? '0'+khMin : khMin}\n🔥 វគ្គ៖ ${sessionName}\n💡 សូមពិនិត្យក្រាហ្វ H1/M15 និងកំណត់ Stop Loss ជានិច្ច!`);
  }

  // 2. Fetch ForexFactory Weekly Calendar JSON (Free Public Feed)
  try {
    const url = "https://nfs.faireconomy.media/ff_calendar_thisweek.json";
    const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (res.getResponseCode() === 200) {
      const events = JSON.parse(res.getContentText());
      events.forEach(ev => {
        if (ev.country === "USD" && ev.impact === "High") {
          const evTime = new Date(ev.date).getTime();
          const diffMins = Math.round((evTime - now.getTime()) / 60000);
          const alertKey = "ALERTED_" + ev.title + "_" + ev.date;

          // Alert 30 mins before Red USD News
          if (diffMins > 0 && diffMins <= 30 && !props.getProperty(alertKey)) {
            sendTgMessage(`🚨 ប្រកាសអាសន្នព័ត៌មានមាស (USD High Impact)!\n📰 ព័ត៌មាន៖ ${ev.title}\n⏳ នៅសល់៖ ${diffMins} នាទីទៀត\n📊 ព្យាករណ៍៖ ${ev.forecast || 'N/A'} | មុន៖ ${ev.previous || 'N/A'}\n🛡️ ទុន $10 សូមផ្អាកចូល Order ឬ បិទ Order សិន ដើម្បីការពារការលោតតម្លៃខ្លាំង!`);
            props.setProperty(alertKey, "1");
          }
        }
      });
    }
  } catch (e) {
    Logger.log("Calendar check error: " + e);
  }
}

function sendTgMessage(text) {
  const url = `https://api.telegram.org/bot${TELEGRAM_TOKEN}/sendMessage`;
  UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({
      chat_id: TELEGRAM_CHAT_ID,
      text: text,
      parse_mode: "HTML"
    })
  });
}
